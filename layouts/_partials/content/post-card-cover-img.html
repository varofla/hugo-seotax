(gt $width 0) -}} {{- if and $preserveRatio (ge .Width $desktop1xWidth) -}} {{- $desktop1x := .Resize (printf "%dx q95" $desktop1xWidth) -}} {{- $src = $desktop1x.RelPermalink -}} {{- $attrs = printf `width="%d" height="%d"` $desktop1x.Width $desktop1x.Height -}} {{- $srcsetParts := slice (printf "%s %dw" $desktop1x.RelPermalink $desktop1x.Width) -}} {{- if ge .Width $desktop2xWidth -}} {{- $desktop2x := .Resize (printf "%dx q95" $desktop2xWidth) -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $desktop2x.RelPermalink $desktop2x.Width) -}} {{- end -}} {{- $srcset = delimit $srcsetParts ", " -}} {{- else if and (not $preserveRatio) (gt $height 0) (ge .Width $width) (ge .Height $height) -}} {{- $desktop1x := .Fill (printf "%dx%d Center q95" $desktop1xWidth $desktop1xHeight) -}} {{- $src = $desktop1x.RelPermalink -}} {{- $attrs = printf `width="%d" height="%d"` $width $height -}} {{- $srcsetParts := slice (printf "%s %dw" $desktop1x.RelPermalink $desktop1x.Width) -}} {{- if and (ge .Width $desktop2xWidth) (ge .Height $desktop2xHeight) -}} {{- $desktop2x := .Fill (printf "%dx%d Center q95" $desktop2xWidth $desktop2xHeight) -}} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $desktop2x.RelPermalink $desktop2x.Width) -}} {{- end -}} {{- $srcset = delimit $srcsetParts ", " -}} {{- else -}} {{- $src = .RelPermalink -}} {{- $attrs = printf `width="%d" height="%d"` .Width .Height -}} {{- end -}} {{- end -}} {{- end -}}
